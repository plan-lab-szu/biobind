cmake_minimum_required(VERSION 3.13)
project(BioBindBGV LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(SEAL 4.1 REQUIRED)
add_executable(bgv-measure bgv-measure.cpp)
target_link_libraries(bgv-measure PRIVATE SEAL::seal)
if(BIOBIND_ZSTD_LIBRARY)
  target_link_libraries(bgv-measure PRIVATE "${BIOBIND_ZSTD_LIBRARY}")
endif()
if(WIN32)
  target_link_libraries(bgv-measure PRIVATE bcrypt)
endif()

find_package(OpenMP REQUIRED)
target_link_libraries(bgv-measure PRIVATE OpenMP::OpenMP_CXX)
